        function update() {
            if (gameState !== GAME_STATES.PLAYING || gameOver) return;

            // Ship follows the cursor / mouse-drag target with smooth
            // damping. Mouse, touch, and the canvas drag handler (which
            // updates targetX on move events) are the only ways to
            // steer — keyboard arrow movement is gone.
            player.x += (player.targetX - player.x) * 0.3;

            if (player.x < player.radius) player.x = player.radius;
            if (player.x > canvas.width - player.radius) player.x = canvas.width - player.radius;

            if (gameMode === GAME_MODES.CLASSIC) {
                // Classic: stars fall; catching them gives points, missing
                // them costs a life.
                if (Math.random() < 0.05) {
                    stars.push(createStar());
                }

                for (let i = stars.length - 1; i >= 0; i--) {
                    const star = stars[i];
                    star.y += star.speed;

                    const dx = player.x - star.x;
                    const dy = player.y - star.y;
                    const distance = Math.sqrt(dx * dx + dy * dy);
                    const collisionDistance = player.radius + star.radius;

                }
            }

        function loadSettings() {
            // Defensive defaults that match the let-settings initialiser
            // above. We return them as a fresh object on every call so
            // mutation by callers (e.g. toggle handlers) never bleeds
            // across accounts.
            // Same defaults as the let-settings initialiser above. 0.7 here
            // means brand-new visitors (no localStorage yet) hear the
            // menu + classic BGM at a comfortable level — the previous
            // 0.3 default was easy to mistake for a broken control since
            // many devices barely audibly play at 30%.
            const defaults = {
                music: true,
                classicMusic: true,
                survivalMusic: true,
                sfx: true,
                menuVolume: 0.7,
                classicVolume: 0.7,
                survivalVolume: 0.7
            };
            try {
                const raw = localStorage.getItem(settingsStorageKey());
                if (raw) {
                    const parsed = JSON.parse(raw);
                    // Migration rules:
                    //   * the music / classicMusic split (old accounts
                    //     used music to silence BOTH tracks) is filled
                    //     in on a per-field basis so the legacy value
                    //     sticks for the few users who relied on it.
                    //   * before this deploy, every volume defaulted to
                    //     0.3. If this account pre-dates the bump (i.e.
                    //     no schemaVersion) AND its stored value matches
                    //     that legacy 0.3 default, push it to the new
                    //     default.  Otherwise the user clearly chose
                    //     this number, so pass it through clampVolume
                    //     untouched.
                    //   * schemaVersion is bumped (and included in the
                    //     return object) so saveSettings persists it;
                    //     future loads skip the migration branch.
                    // Bumped migration threshold from <2 → <3 so accounts
                    // that saved during the previous deploy (which only
                    // had music + classicMusic) get the new survivalMusic
                    // + survivalVolume fields added with the same 0.3→0.7
                    // bump rule we used for the menu / classic volumes.
                    const needsMigration = (parsed.schemaVersion || 0) < 3;
                    const musicVal = typeof parsed.music === 'boolean' ? parsed.music : defaults.music;
                    const result = {
                        schemaVersion: 3,                  // always current after migration runs
                        music: musicVal,
                        classicMusic: parsed.classicMusic ?? musicVal,
                        survivalMusic: parsed.survivalMusic ?? musicVal,
                        sfx: typeof parsed.sfx === 'boolean' ? parsed.sfx : defaults.sfx,
                        menuVolume: (needsMigration && clampVolume(parsed.menuVolume) === 0.3)
                            ? defaults.menuVolume
                            : clampVolume(parsed.menuVolume),
                        classicVolume: (needsMigration && clampVolume(parsed.classicVolume) === 0.3)
                            ? defaults.classicVolume
                            : clampVolume(parsed.classicVolume),
                        survivalVolume: (needsMigration && clampVolume(parsed.survivalVolume) === 0.3)
                            ? defaults.survivalVolume
                            : clampVolume(parsed.survivalVolume)
                    };
                    if (needsMigration) {
                        try { localStorage.setItem(settingsStorageKey(), JSON.stringify(result)); } catch (e) {}
                    }
                    return result;
                }
            } catch (e) { /* fall through to defaults */ }
            return defaults;
        }
