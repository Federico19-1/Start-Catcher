                        playClickSound();
                        recordMissionEvent('toggled_setting');
                    }
                    break;
                case 'settingsBack':
                    if (gameStateBeforeSettings === GAME_STATES.PAUSED) {
                        gameState = GAME_STATES.PAUSED;
                    } else {
                        gameState = GAME_STATES.MENU;
                        showMobileInput(false);
                    }
                    gameStateBeforeSettings = GAME_STATES.MENU;
                    break;
                default:
                    break;
            }
