                    }
                }
            }

            // ----- Battle column (right side of the new 3-column grid) -----
            // Mirrors the Survival column rendering exactly but reads
            // from the `${user}__battle` bucket via getBattleRecords
            // and uses a warm orange palette ('🎯 BATTLE', 'Enemies killed',
            // '#ff8a3d' borders) to distinguish it from the two cold-toned
            // siblings. Records are rendered as 'N kills' instead of 'Ns'.
            ctx.save();
            const battleRecords = currentUser ? getBattleRecords(currentUser) : [];
            ctx.shadowBlur = 18;
            ctx.shadowColor = '#ff8a3d';
            ctx.fillStyle = 'rgba(74, 26, 10, 0.40)';
            ctx.beginPath();
            ctx.roundRect(col3X, colY, colW, colH, 12);
            ctx.fill();
            ctx.shadowBlur = 0;
            ctx.strokeStyle = 'rgba(255, 138, 61, 0.6)';
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.roundRect(col3X, colY, colW, colH, 12);
            ctx.stroke();
            // Header
            ctx.textAlign = 'center';
            ctx.font = 'bold 22px Orbitron, sans-serif';
            ctx.fillStyle = '#ffd2a8';
            ctx.shadowBlur = 12;
            ctx.shadowColor = '#ff8a3d';
            ctx.fillText('🎯 BATTLE', col3X + colW / 2, colY + 32);
            ctx.shadowBlur = 0;
            ctx.font = '13px Orbitron, sans-serif';
            ctx.fillStyle = '#9aa6b2';
            ctx.fillText('Enemies killed', col3X + colW / 2, colY + 52);
            // Body
            const hasBattle = battleRecords.some(r => r > 0);
            if (!hasBattle) {
                ctx.font = 'bold 14px Orbitron, sans-serif';
                ctx.fillStyle = '#ffd2a8';
                ctx.fillText('No records yet', col3X + colW / 2, colY + 110);
            } else {
                ctx.font = '15px Orbitron, sans-serif';
                battleRecords.slice(0, 5).forEach((record, idx) => {
                    if (record <= 0) return;
                    let label, color, weight = 'normal';
                    if (idx === 0)      { label = '🥇 ' + record; color = '#ffd700'; weight = 'bold'; }
                    else if (idx === 1) { label = '🥈 ' + record; color = '#c0c0c0'; weight = 'bold'; }
                    else if (idx === 2) { label = '🥉 ' + record; color = '#cd7f32'; weight = 'bold'; }
                    else                { label = (idx + 1) + '. ' + record; color = '#ffb59c'; }
                    ctx.font = (weight === 'bold' ? 'bold ' : '') + '15px Orbitron, sans-serif';
                    ctx.fillStyle = color;
                    ctx.fillText(label + ' kills', col3X + colW / 2, colY + 90 + idx * 28);
                });
            }
            ctx.textAlign = 'left';
            ctx.restore();

            // ----- "How much time did you play" card -----
            // Sits beneath the two record columns; the value is the
            // per-account cumulative ms spent in the PLAYING state,
            // formatted nicely. Always rendered (zero value still
            // shows "0s" so the screen never feels broken on a
            // brand-new account).
            const cardX = 75;
            const cardY = 425;
            const cardW = canvas.width - cardX * 2;
            const cardH = 90;
            ctx.shadowBlur = 22;
            ctx.shadowColor = '#66e0ff';
            const cardGrad = ctx.createLinearGradient(cardX, cardY, cardX, cardY + cardH);
            cardGrad.addColorStop(0, 'rgba(20, 50, 80, 0.85)');
            cardGrad.addColorStop(1, 'rgba(10, 25, 50, 0.95)');
            ctx.fillStyle = cardGrad;
            ctx.beginPath();
            ctx.roundRect(cardX, cardY, cardW, cardH, 14);
            ctx.fill();
            ctx.shadowBlur = 0;
            ctx.strokeStyle = 'rgba(102, 224, 255, 0.55)';
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.roundRect(cardX, cardY, cardW, cardH, 14);
            ctx.stroke();

            // Header label (left)
            ctx.textAlign = 'left';
            ctx.font = 'bold 13px Orbitron, sans-serif';
            ctx.shadowColor = 'black';
            ctx.shadowBlur = 4;
            ctx.fillStyle = '#9cd3ff';
            ctx.fillText('⏱  HOW MUCH TIME DID YOU PLAY', cardX + 22, cardY + 26);

            // Friendly description
            ctx.font = '11px Orbitron, sans-serif';
            ctx.fillStyle = '#7eaad6';
            ctx.shadowBlur = 3;
            ctx.fillText('Active gameplay across this account', cardX + 22, cardY + 46);

            // Big value (right-aligned)
            const playMs = currentUser ? getPlayTime(currentUser) : 0;
            const playText = formatPlayTime(playMs);
            ctx.textAlign = 'right';
            ctx.font = 'bold 30px Orbitron, sans-serif';
            ctx.shadowColor = '#66e0ff';
            ctx.shadowBlur = 18;
            ctx.fillStyle = '#ffffff';
            ctx.fillText(playText, cardX + cardW - 22, cardY + 60);
            ctx.shadowBlur = 0;

            // Lightweight sub-label under the value
            ctx.font = '11px Orbitron, sans-serif';
            ctx.fillStyle = '#7eaad6';
            ctx.shadowBlur = 3;
            ctx.fillText('h   m   s', cardX + cardW - 22, cardY + 78);

            const backY = 540;
            const backW = 160;
            const backH = 35;
            const backX = canvas.width / 2 - backW / 2;
            ctx.shadowBlur = 10;
            ctx.shadowColor = '#4af';
            ctx.fillStyle = '#2a4a7a';
            ctx.beginPath();
            ctx.roundRect(backX, backY, backW, backH, 10);
            ctx.fill();
            ctx.shadowBlur = 0;
